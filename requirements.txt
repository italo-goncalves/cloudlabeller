# CloudLabeller dependencies — the authoritative, commented list; pyproject.toml
# mirrors it. The desktop app's Environment installs the pinned lock made from
# these (apps/desktop/environment/requirements.lock, tools/lock_environment.py).
#
# Dev env (the Engine, the Jobs and the tests):  pip install -e .[dev]

# Core
numpy>=1.24
scipy>=1.10
# Headless: the app never opens an OpenCV window. Don't install opencv-python beside it: both own cv2/.
opencv-python-headless>=4.8
pillow>=10          # polygon rasterisation (core/raster.py)

# Geometry / IO — PLY is read by trimesh and written by io/ply.py (no plyfile).
trimesh>=4.0
laspy>=2.5
pyproj>=3.6          # CRS-aware export (ENU -> projected EPSG)

# Photogrammetry — written against the pycolmap 4.2 API. Both packages provide
# the same `pycolmap` module: the CPU build on Windows (GPU SIFT/MVS there go
# through colmap.exe), the CUDA 12 build on Linux/WSL (GPU SIFT in-process).
pycolmap==4.2.*; sys_platform == "win32"
pycolmap-cuda12==4.2.*; sys_platform == "linux"

# ML — Ítalo's U-Net is TensorFlow/Keras. Pin 2.19: 2.20+ wheels failed to
# load on this machine. NOTE: importing TF needs the MSVC-runtime preload in
# cloudlabeller/__init__.py (Anaconda ships a 2020 msvcp140.dll that otherwise
# shadows System32's and breaks the TF DLL init). Native-Windows TF is
# CPU-only since 2.11; GPU training runs in WSL, with the CUDA wheels.
tensorflow==2.19.*
tensorflow[and-cuda]==2.19.*; sys_platform == "linux"
scikit-image>=0.22   # augmentation (gamma correction) in ml/dataset_builder.py
